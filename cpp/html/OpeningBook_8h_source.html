72 [[nodiscard]] value_t binarySearch(const key_t& huffmanCode) const {
73 // one could also use: std::lower_bound()
74 int l = 0; // dont use size_t to prevent undesired underflows
75 int r = m_book.size() - 1;
76 while (r >= l) {
77 const auto mid = (l + r + 1) / 2;
78 auto p = m_book.at(mid);
79 if (std::get<0>(p) == huffmanCode) {
80 return std::get<1>(p); // Found! return the value for this position
81 }
82 if (std::get<0>(p) > huffmanCode) {
83 r = mid - 1;
84 } else { // p < huffmanCode
85 l = mid + 1;
86 }
87 }
88
89 // Nothing found:
90 return std::numeric_limits<value_t>::min();
91 }

214 static std::tuple<key_t, int> readline(std::ifstream& file,
215 const bool with_distances,
216 const bool is_8ply) {
217 const decltype(file.gcount()) bytes_position = is_8ply ? 3 : 4;
218 char buffer[4] = {}; // Max buffer size for reading
219 file.read(buffer, bytes_position);
220
221 if (file.gcount() != bytes_position) {
222 // EOF or read error
223 return {0, 0};
224 }
225
226 // Convert the read bytes into an integer
227 key_t huffman_position = 0;
228 for (decltype(file.gcount()) i = 0; i < bytes_position; ++i) {
229 huffman_position =
230 (huffman_position << 8) | static_cast<unsigned char>(buffer[i]);
231 }
232
233 if (!is_8ply) {
234 // Handle signed interpretation for 4-byte numbers
235 if (huffman_position & (1LL << ((bytes_position * 8) - 1))) {
236 huffman_position -= (1LL << (bytes_position * 8));
237 }
238 }
239
240 value_t score = 0;
241 if (with_distances) {
242 // Read one additional byte for the score
243 char score_byte;
244 if (file.read(&score_byte, 1)) {
245 score = static_cast<int8_t>(score_byte);
246 } else {
247 // EOF after reading huffman_position
248 return {0, 0};
249 }
250 } else {
251 // Last 2 bits indicate the score
252 score = (static_cast<value_t>(huffman_position) & 3) * -1;
253 huffman_position = huffman_position & ~3;
254 }
255
256 return {huffman_position, score};
257 }

271 static std::vector<std::tuple<key_t, value_t>> readBook(
272 const std::filesystem::path& filename, const bool with_distances = true,
273 const bool is_8ply = false) {
274 std::vector<std::tuple<key_t, value_t>> book; // To store the book entries
275 std::ifstream file(filename, std::ios::binary);
276 if (!file) {
277 std::cerr << "Failed to open file: " << filename.string() << '\n';
278 return book; // Return an empty book if the file can't be opened
279 }
280
281 while (true) {
282 auto [position, score] = readline(file, with_distances, is_8ply);
283 if (file.eof()) {
284 break; // End of file reached
285 }
286 book.emplace_back(position, score);
287 }
288
289 return book;
290 }

350 [[nodiscard]] int getBoardValue(const Board& b) const {
351 if (!((m_is8ply && b.countTokens() == 8) || b.countTokens() == 12)) {
352 return NONE_VALUE;
353 }
354
355 // # first try this position
356 auto p = b.toHuffman();
357 int val = binarySearch(p);
358 if (val != NONE_VALUE) {
359 return convertValue(val, b);
360 }
361
362 // # Did not find position. Look for the mirrored equivalent
363 p = b.mirror().toHuffman();
364 val = binarySearch(p);
365 if (!m_withDistances && val == NONE_VALUE) {
366 // only for the 8-ply and 12-ply database without distances
367 val = 1; // if a position is not in the database, then this means that
368 // player 1 wins
369
370 // obsolete:
371 // Apparently, positions with 2 immediate threats for player Red are
372 // missing in the 8-ply database
373 // if (m_is8ply && !b.generateNonLosingMoves()) {
374 // val = -1;
375 //}
376 } else if (val == NONE_VALUE) {
377 // This is a special case. Positions, where player 1 (yellow) can
378 // immediately win, are not encoded in the databases.
379 return (b.movesLeft() + 1) / 2;
380 }
381 assert(val != NONE_VALUE);
