102 static int scoreToMovesLeft(const int score, const Board& b) noexcept {
103 if (score == 0) {
104 return b.movesLeft();
105 }
106 const int p = (b.movesLeft() + 1) % 2; // 1 -> yellow, 0 -> red
107 const int sgnScore = score < 0 ? 1 : 0;
108 const int absScore = score < 0 ? -score : score;
109 const int mvFinalLeft = 2 * (absScore - 1) + (sgnScore ^ p);
110 return b.movesLeft() - mvFinalLeft;
111 }

126 static int rollout(Board b) noexcept {
127 int ply = 0;
128
129 while (true) {
130 if (b.canWin()) {
131 const int score = (b.movesLeft() + 1) / 2;
132 return (ply % 2 == 0) ? score : -score;
133 }
134
135 if (!b.movesLeft()) {
136 return 0;
137 }
138
139 auto moves = b.generateNonLosingMoves();
140 if (!moves) {
141 const int score = -(b.movesLeft() / 2);
142 return (ply % 2 == 0) ? score : -score;
143 }
144
145 const auto mv = Board::nextMove(moves);

150 // playMoveFast() method which does not check for legality of the move
151 // since we already know that it is legal).
152 b = b.playBitMaskOnCopy(mv);
153 ply++;

177 int mtdf(const Board& b, const int firstGuess,
178 const int maxDepth = -1) noexcept {
179 // MTD(f) algorithm by Aske Plaat: Plaat, Aske; Jonathan Schaeffer; Wim
180 // Pijls; Arie de Bruin (November 1996). "Best-first Fixed-depth Minimax
181 // Algorithms". Artificial Intelligence. 87 (1–2): 255–293.
182 // doi:10.1016/0004-3702(95)00126-3
183 auto g = firstGuess;
184 int upperBound = INT32_MAX;
185 int lowerBound = INT32_MIN;
186
187 while (lowerBound < upperBound) {
188 const auto beta = std::max(g, lowerBound + 1);
189 g = negamax(b, beta - 1, beta, 0, maxDepth);
190 if (g < beta) {
191 upperBound = g;
192 } else {
193 lowerBound = g;
194 }
195 }
196 return g;
197 }

210 int nullWindow(const Board& b, const int maxDepth = -1) noexcept {
211 int min = -b.movesLeft() / 2;
212 int max = (b.movesLeft() + 1) / 2;
213
214 while (min < max) {
215 int mid = min + (max - min) / 2;
216 if (mid <= 0 && min / 2 < mid)
217 mid = min / 2;
218 else if (mid >= 0 && max / 2 > mid)
219 mid = max / 2;
220 int r = negamax(b, mid, mid + 1, 0, maxDepth);
221 if (r <= mid) {
222 max = r;
223 } else {
224 min = r;
225 }
226 }
227 return min;
228 }
